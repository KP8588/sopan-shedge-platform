class _LoginScreenState extends State<LoginScreen> {
  final _formKey = GlobalKey<FormState>();
  final _username = TextEditingController();
  final _password = TextEditingController();
  bool _loading = false;
  bool _hidePassword = true;
  String? _error;

  @override
  void dispose() {
    _username.dispose();
    _password.dispose();
    super.dispose();
  }

  
Future<void> _signIn() async {
  if (_loading) return;

  if (!_formKey.currentState!.validate()) return;

  FocusScope.of(context).unfocus();

  setState(() {
    _loading = true;
    _error = null;
  });

  debugPrint('LOGIN UI: Login button pressed');

  try {
    debugPrint('LOGIN UI: Calling ApiService.login');

    await ApiService.login(
      username: _username.text,
      password: _password.text,
    );

    debugPrint('LOGIN UI: API login succeeded');

    if (!mounted) return;

    Navigator.of(context).pushReplacement(
      MaterialPageRoute(
        builder: (_) => const StudentDashboard(),
      ),
    );
  } catch (e) {
    // Never log credentials or authentication tokens.
    debugPrint('LOGIN UI: Failed with ${e.runtimeType}');
    debugPrint('LOGIN UI: Error = $e');

    if (!mounted) return;

    setState(() {
      _error = e.toString().replaceFirst('Exception: ', '');
    });
  } finally {
    if (mounted) {
      setState(() {
        _loading = false;
      });
    }
  }
}

class _CourseListScreenState extends State<CourseListScreen> {
  late Future<List<Map<String, dynamic>>> _future;

  @override
  void initState() {
    super.initState();
    _future = ApiService.getCourses();
  }

  void _reload() => setState(() => _future = ApiService.getCourses());

  Future<void> _refresh() async {
    final next = ApiService.getCourses();
    setState(() => _future = next);
    await next;
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('My Courses'),
        actions: [
          IconButton(onPressed: _reload, icon: const Icon(Icons.refresh)),
        ],
      ),
      body: RefreshIndicator(
        onRefresh: _refresh,
        child: FutureBuilder<List<Map<String, dynamic>>>(
          future: _future,
          builder: (context, snapshot) {
            if (snapshot.connectionState == ConnectionState.waiting) {
              return const _StatusList(
                icon: Icons.hourglass_top,
                title: 'Loading courses...',
              );
            }
            if (snapshot.hasError) {
              return _StatusList(
                icon: Icons.cloud_off,
                title: 'Unable to load courses',
                message: '${snapshot.error}',
                actionLabel: 'Try Again',
                onAction: _reload,
              );
            }
            final courses = snapshot.data ?? [];
            if (courses.isEmpty) {
              return const _StatusList(
                icon: Icons.menu_book_outlined,
                title: 'No active courses available yet.',
                message: 'Courses will appear here when available.',
              );

            return ListView.builder(
              padding: const EdgeInsets.all(16),
              itemCount: courses.length,
              itemBuilder: (context, index) {
                final course = courses[index];
                final title = (course['title'] ?? 'Course ${index + 1}')

                final description = (course['description'] ?? '').toString();

class LiveClassesScreen extends StatefulWidget {
  const LiveClassesScreen({super.key});

  @override
  State<LiveClassesScreen> createState() => _LiveClassesScreenState();
}

class _LiveClassesScreenState extends State<LiveClassesScreen> {
  late Future<List<Map<String, dynamic>>> _future;

  @override
  void initState() {
    super.initState();
    _future = ApiService.getClassrooms();
  }

  void _reload() => setState(() => _future = ApiService.getClassrooms());

  Future<void> _refresh() async {
    final next = ApiService.getClassrooms();
    setState(() => _future = next);
    await next;
  }

  String _status(Map<String, dynamic> item) {
    if (item['is_active'] != true) return 'Inactive';
    final start = DateTime.tryParse(
      item['scheduled_at']?.toString() ?? '',
    )?.toLocal();
    if (start == null) return 'Schedule unavailable';

    final duration =
        int.tryParse(item['duration_minutes']?.toString() ?? '') ?? 60;
    final end = start.add(Duration(minutes: duration));
    final now = DateTime.now();
    if (now.isBefore(start)) return 'Upcoming';
    if (now.isBefore(end)) return 'In progress';
    return 'Past';
  }

  String _formatDate(dynamic raw) {
    final date = DateTime.tryParse(raw?.toString() ?? '')?.toLocal();
    if (date == null) return 'Schedule unavailable';
    final day = date.day.toString().padLeft(2, '0');
    final month = date.month.toString().padLeft(2, '0');
    final hour = date.hour % 12 == 0 ? 12 : date.hour % 12;
    final minute = date.minute.toString().padLeft(2, '0');
    final period = date.hour >= 12 ? 'PM' : 'AM';
    return '$day/$month/${date.year} at $hour:$minute $period';
  }

  Future<void> _openMeeting(Map<String, dynamic> item) async {
    final status = _status(item);

    if (status != 'Upcoming' && status != 'In progress') {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text('This class is not currently joinable.'),
        ),
      );
      return;
    }

    await Navigator.of(context).push(
      MaterialPageRoute<void>(
        builder: (_) => ClassroomPreviewScreen(classroom: item),
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Live Classes'),
        actions: [
          IconButton(
            tooltip: 'Refresh classes',
            onPressed: _reload,
            icon: const Icon(Icons.refresh),
          ),
        ],
      ),
      body: RefreshIndicator(
        onRefresh: _refresh,
        child: FutureBuilder<List<Map<String, dynamic>>>(
          future: _future,
          builder: (context, snapshot) {
            if (snapshot.connectionState == ConnectionState.waiting) {
              return const _StatusList(
                icon: Icons.hourglass_top,
                title: 'Loading live classes...',
              );
            }
            if (snapshot.hasError) {
              return _StatusList(
                icon: Icons.cloud_off,
                title: 'Unable to load live classes',
                message: '${snapshot.error}',
                actionLabel: 'Try Again',
                onAction: _reload,
              );
            }

            final classes = [...?snapshot.data]
              ..sort((a, b) {
                final first =
                    DateTime.tryParse(a['scheduled_at']?.toString() ?? '') ??
                    DateTime(2100);
                final second =
                    DateTime.tryParse(b['scheduled_at']?.toString() ?? '') ??
                    DateTime(2100);
                return first.compareTo(second);
              });

            if (classes.isEmpty) {
              return const _StatusList(
                icon: Icons.video_camera_front_outlined,
                title: 'No live classes available',

          '${response.body}',
        );

      final serverUrl = data['serverUrl']?.toString() ?? '';
      final roomToken = data['token']?.toString() ?? '';

      if (serverUrl.isEmpty || roomToken.isEmpty) {
        throw Exception('Django returned incomplete LiveKit credentials.');

debugPrint('LIVEKIT URL scheme: ${Uri.tryParse(serverUrl)?.scheme}');
debugPrint('LIVEKIT URL host: ${Uri.tryParse(serverUrl)?.host}');

import 'dart:async';
import 'dart:convert';

import 'package:flutter/foundation.dart';
import 'package:http/http.dart' as http;

class ApiService {
  ApiService._();

  // Development URL: the laptop's current IPv4 address.
  // The Android phone must be able to reach this address.
  // static const String baseUrl = 'http://10.84.104.230:8000';
  static const String baseUrl = 'http://127.0.0.1:8000';

  static const Duration _requestTimeout = Duration(seconds: 15);

  static String? authToken;
  static Map<String, dynamic>? currentUser;

  static Future<Map<String, dynamic>> login({
    required String username,
    required String password,
  }) async {
    final uri = Uri.parse('$baseUrl/api/auth/login/');

    debugPrint('LOGIN: Starting request');
    debugPrint('LOGIN: URL = $uri');

    try {
      final response = await http
          .post(
            uri,
            headers: const {
              'Accept': 'application/json',
              'Content-Type': 'application/json',
            },
            body: jsonEncode({
              'username': username.trim(),
              'password': password,
            }),
          )
          .timeout(_requestTimeout);

      debugPrint('LOGIN: HTTP ${response.statusCode}');

      final dynamic decoded;

      try {
        decoded = jsonDecode(response.body);
      } on FormatException {
        debugPrint('LOGIN: Server returned invalid JSON');
        throw Exception('The server returned an invalid response.');
      }

      if (decoded is! Map) {
        throw Exception('Unexpected login response from the server.');
      }

      final data = Map<String, dynamic>.from(decoded);

      if (response.statusCode >= 200 &&
          response.statusCode < 300 &&
          data['success'] == true &&
          data['token'] != null &&
          data['token'].toString().isNotEmpty) {
        authToken = data['token'].toString();

        final user = data['user'];
        currentUser = user is Map
            ? Map<String, dynamic>.from(user)
            : <String, dynamic>{'username': username.trim()};

        debugPrint('LOGIN: Authentication successful');
        return data;
      }

      // Do not print the response token or password.
      final message = _errorMessage(data, 'Login failed.');
      debugPrint('LOGIN: Rejected with HTTP ${response.statusCode}');

      throw Exception(message);
    } on TimeoutException {
      debugPrint('LOGIN: Request timed out after 15 seconds');
      throw Exception(
        'Login timed out. Check the phone connection to the backend.',
      );
    } on http.ClientException catch (e) {
      debugPrint('LOGIN: Network client error: $e');
      throw Exception('Could not connect to the Django server.');
    } on FormatException {
      debugPrint('LOGIN: Response parsing failed');
      throw Exception('The server returned invalid data.');
    } catch (e) {
      if (e is Exception) {
        rethrow;
      }

      debugPrint('LOGIN: Unexpected error: ${e.runtimeType}');
      throw Exception('An unexpected error occurred during login.');
    }
  }

  static Future<Map<String, dynamic>> getLiveKitToken(int classroomId) async {
    if (classroomId <= 0) {
      throw ArgumentError.value(
        classroomId,
        'classroomId',
        'Classroom ID must be positive.',

          .timeout(_requestTimeout);

      debugPrint('LIVEKIT: HTTP ${response.statusCode}');

      final dynamic decoded;
      try {
        decoded = jsonDecode(response.body);
      } on FormatException {
        throw Exception('Django returned an invalid LiveKit response.');
      }

      if (decoded is! Map) {
        throw Exception('Unexpected LiveKit response from Django.');
      }

      final data = Map<String, dynamic>.from(decoded);

      if (response.statusCode < 200 || response.statusCode >= 300) {
        throw Exception(_errorMessage(data, 'Could not join this classroom.'));
      }

      final serverUrl = data['serverUrl'];
      final accessToken = data['token'];
      final roomName = data['roomName'];
      final isTrainer = data['isTrainer'];

      if (serverUrl is! String ||

  static Future<List<Map<String, dynamic>>> getCourses() async {
    final uri = Uri.parse('$baseUrl/api/courses/');

    try {
      final response = await http
          .get(uri, headers: _headers())
          .timeout(_requestTimeout);

      debugPrint('COURSES: HTTP ${response.statusCode}');

      if (response.statusCode != 200) {
        throw Exception(
          'Could not load courses (HTTP ${response.statusCode}).',
        );
      }

      final dynamic decoded = jsonDecode(response.body);
      final List<dynamic> results;

      if (decoded is List) {
        results = decoded;
      } else if (decoded is Map && decoded['results'] is List) {
        results = decoded['results'] as List<dynamic>;
      } else {
        throw Exception('Unexpected course data received from Django.');
      }

      return results
          .whereType<Map>()
          .map((item) => Map<String, dynamic>.from(item))
          .where((item) => item['is_active'] != false)
          .toList();
    } on TimeoutException {
      throw Exception('Loading courses timed out. Please try again.');
    } on http.ClientException {
      throw Exception('Could not connect to Django while loading courses.');
    } on FormatException {
      throw Exception('Django returned invalid course data.');
    }
  }

  static Future<List<Map<String, dynamic>>> getClassrooms() async {
    final uri = Uri.parse('$baseUrl/api/classrooms/');

    try {
      final response = await http
          .get(uri, headers: _headers())
          .timeout(_requestTimeout);

      debugPrint('CLASSROOMS: HTTP ${response.statusCode}');

      if (response.statusCode != 200) {
        throw Exception(
          'Could not load live classes (HTTP ${response.statusCode}).',
        );
      }

      final dynamic decoded = jsonDecode(response.body);

      if (decoded is List) {
        return decoded
            .whereType<Map>()
            .map((item) => Map<String, dynamic>.from(item))
            .toList();
      }

      if (decoded is Map && decoded['results'] is List) {
        return (decoded['results'] as List)
            .whereType<Map>()
            .map((item) => Map<String, dynamic>.from(item))
            .toList();
      }

      if (decoded is Map && decoded.containsKey('id')) {
        return [Map<String, dynamic>.from(decoded)];
      }

      throw Exception('Unexpected classroom data received from Django.');
    } on TimeoutException {
      throw Exception('Loading live classes timed out. Please try again.');
    } on http.ClientException {
      throw Exception('Could not connect to Django while loading classes.');
    } on FormatException {
      throw Exception('Django returned invalid classroom data.');
    }
  }

  static Map<String, String> _headers() {
    final headers = <String, String>{'Accept': 'application/json'};

    final token = authToken;
    if (token != null && token.isNotEmpty) {
      headers['Authorization'] = 'Token $token';
    }

    return headers;
  }

  static void logout() {
    authToken = null;
    currentUser = null;
  }

  static String _errorMessage(Map<String, dynamic> data, String fallback) {
    for (final key in ['detail', 'message', 'error']) {
      final value = data[key];

      if (value is String && value.trim().isNotEmpty) {
        return value;
      }
    }

    final nonFieldErrors = data['non_field_errors'];
    if (nonFieldErrors is List && nonFieldErrors.isNotEmpty) {
      return nonFieldErrors.join(' ');
    }

    return fallback;
  }

import 'dart:convert';
import 'dart:async';

import 'package:web_socket_channel/io.dart';
import 'package:web_socket_channel/status.dart' as ws_status;

class ClassroomSignalingService {
  IOWebSocketChannel? _channel;
  StreamSubscription<dynamic>? _subscription;

  bool get isConnected => _channel != null;

  Future<void> connect({
    required String baseUrl,
    required String token,
    required int classroomId,
    required void Function(Map<String, dynamic>) onMessage,
    required void Function(String) onStatus,
  }) async {
    await disconnect();

    if (token.trim().isEmpty) {
      onStatus('Not connected: please log in again.');
      return;
    }

    final httpUri = Uri.parse(baseUrl);
    final scheme = httpUri.scheme == 'https' ? 'wss' : 'ws';
    final uri = Uri(
      scheme: scheme,
      host: httpUri.host,
      port: httpUri.hasPort ? httpUri.port : null,
      path: '/ws/classrooms/$classroomId/',
    );

    try {
      onStatus('Connecting to classroom signaling…');

      final channel = IOWebSocketChannel.connect(
        uri,
        headers: {'Authorization': 'Token $token'},
        connectTimeout: const Duration(seconds: 10),
      );

      _channel = channel;

      await channel.ready.timeout(const Duration(seconds: 12));
      onStatus('Signaling connected');

      _subscription = channel.stream.listen(
        (event) {
          try {
            if (event is String) {
              final decoded = jsonDecode(event);
              if (decoded is Map<String, dynamic>) {
                onMessage(decoded);
              }
            }
          } catch (_) {
            onStatus('Received an invalid signaling message.');
          }
        },
        onError: (Object error) {
          onStatus('Signaling connection error: $error');
        },
        onDone: () {
          _channel = null;
          onStatus('Signaling disconnected');
        },
        cancelOnError: false,
      );
    } catch (e) {
      _channel = null;
      onStatus('Could not connect to signaling: $e');
    }
  }

  void send(Map<String, dynamic> message) {
    final channel = _channel;
    if (channel == null) {
      throw StateError('Classroom signaling is not connected.');
    }
    channel.sink.add(jsonEncode(message));
  }

  Future<void> disconnect() async {
    await _subscription?.cancel();
    _subscription = null;

    final channel = _channel;
    _channel = null;

    if (channel != null) {
      await channel.sink.close(ws_status.normalClosure);
    }
  }
}
